通用单链表结点：

```cpp
typedef int ElemType;

struct LNode {
    ElemType data;
    LNode *next;
};

using LinkList = LNode*;
```

### 01 删除带头结点单链表中所有值为 `x` 的结点

```cpp
void DeleteX(LinkList L, ElemType x) {
    LNode *pre = L;
    LNode *p = L->next;

    while (p != nullptr) {
        if (p->data == x) {
            pre->next = p->next;
            delete p;
            p = pre->next;
        } else {
            pre = p;
            p = p->next;
        }
    }
}
```

复杂度：时间 `O(n)`，空间 `O(1)`。

### 02 删除带头结点单链表中的最小值结点

```cpp
bool DeleteMin(LinkList L, ElemType &e) {
    if (L->next == nullptr) return false;

    LNode *pre = L;
    LNode *p = L->next;
    LNode *minPre = L;
    LNode *minP = L->next;

    while (p != nullptr) {
        if (p->data < minP->data) {
            minP = p;
            minPre = pre;
        }
        pre = p;
        p = p->next;
    }

    e = minP->data;
    minPre->next = minP->next;
    delete minP;
    return true;
}
```

复杂度：时间 `O(n)`，空间 `O(1)`。

### 03 带头结点单链表就地逆置

采用头插法重建链表。

```cpp
void Reverse(LinkList L) {
    LNode *p = L->next;
    L->next = nullptr;

    while (p != nullptr) {
        LNode *q = p->next;
        p->next = L->next;
        L->next = p;
        p = q;
    }
}
```

复杂度：时间 `O(n)`，空间 `O(1)`。

### 04 删除值处于给定两值之间的结点

下面按闭区间 `[s,t]` 处理；若题目要求开区间，只需把条件改为 `p->data > s && p->data < t`。

```cpp
void DeleteRange(LinkList L, ElemType s, ElemType t) {
    if (s > t) {
        ElemType tmp = s;
        s = t;
        t = tmp;
    }

    LNode *pre = L;
    LNode *p = L->next;

    while (p != nullptr) {
        if (p->data >= s && p->data <= t) {
            pre->next = p->next;
            delete p;
            p = pre->next;
        } else {
            pre = p;
            p = p->next;
        }
    }
}
```

复杂度：时间 `O(n)`，空间 `O(1)`。

### 05 找两个单链表的公共结点

这里的公共结点指**地址相同的结点**，不是数据值相同。

算法思想：

1. 分别求两个链表长度 `lenA` 和 `lenB`。
2. 让较长链表的指针先走 `abs(lenA-lenB)` 步。
3. 两个指针同时后移，第一次相等的位置就是第一个公共结点。
4. 若同时到达空指针仍不相等，则无公共结点。

复杂度：时间 `O(m+n)`，空间 `O(1)`。

### 06 将 `C={a1,b1,a2,b2,...,an,bn}` 拆成 `A={a1,...,an}` 和 `B={bn,...,b1}`

`A` 用尾插法保持原顺序，`B` 用头插法得到逆序。

```cpp
void Split(LinkList C, LinkList &A, LinkList &B) {
    A = C;
    B = new LNode;
    B->next = nullptr;

    LNode *p = C->next;
    LNode *ra = A;
    A->next = nullptr;

    while (p != nullptr) {
        LNode *q = p->next;
        ra->next = p;
        ra = p;
        p = q;

        if (p != nullptr) {
            q = p->next;
            p->next = B->next;
            B->next = p;
            p = q;
        }
    }
    ra->next = nullptr;
}
```

复杂度：时间 `O(n)`，空间 `O(1)`，不计新建头结点。

### 07 删除递增有序单链表中的重复元素

```cpp
void DeleteDuplicate(LinkList L) {
    LNode *p = L->next;

    while (p != nullptr && p->next != nullptr) {
        if (p->data == p->next->data) {
            LNode *q = p->next;
            p->next = q->next;
            delete q;
        } else {
            p = p->next;
        }
    }
}
```

复杂度：时间 `O(n)`，空间 `O(1)`。

### 08 由两个递增有序链表的公共元素建立新链表 `C`

不破坏 `A`、`B`，因此公共元素要新建结点。

```cpp
LinkList CommonToNewList(LinkList A, LinkList B) {
    LinkList C = new LNode;
    C->next = nullptr;
    LNode *r = C;
    LNode *p = A->next;
    LNode *q = B->next;

    while (p != nullptr && q != nullptr) {
        if (p->data < q->data) {
            p = p->next;
        } else if (p->data > q->data) {
            q = q->next;
        } else {
            LNode *s = new LNode{p->data, nullptr};
            r->next = s;
            r = s;
            p = p->next;
            q = q->next;
        }
    }
    return C;
}
```

复杂度：时间 `O(m+n)`，空间 `O(k)`，`k` 为公共元素个数。

### 09 求两个递增集合链表的交集，并存放到 `A`

```cpp
void IntersectionInA(LinkList A, LinkList B) {
    LNode *pre = A;
    LNode *p = A->next;
    LNode *q = B->next;

    while (p != nullptr && q != nullptr) {
        if (p->data < q->data) {
            pre->next = p->next;
            delete p;
            p = pre->next;
        } else if (p->data > q->data) {
            q = q->next;
        } else {
            pre = p;
            p = p->next;
            q = q->next;
        }
    }

    while (p != nullptr) {
        pre->next = p->next;
        delete p;
        p = pre->next;
    }
}
```

复杂度：时间 `O(m+n)`，空间 `O(1)`。

### 10 判断序列 `B` 是否为序列 `A` 的连续子序列

```cpp
bool IsSubSequence(LinkList A, LinkList B) {
    if (B->next == nullptr) return true;

    for (LNode *pa = A->next; pa != nullptr; pa = pa->next) {
        LNode *p = pa;
        LNode *q = B->next;

        while (p != nullptr && q != nullptr && p->data == q->data) {
            p = p->next;
            q = q->next;
        }

        if (q == nullptr) return true;
    }
    return false;
}
```

复杂度：最坏时间 `O(mn)`，空间 `O(1)`。

### 11 判断带头结点的循环双链表是否对称

```cpp
struct DNode {
    ElemType data;
    DNode *prior;
    DNode *next;
};

using DLinkList = DNode*;

bool Symmetry(DLinkList L) {
    DNode *p = L->next;
    DNode *q = L->prior;

    while (p != q && p->prior != q) {
        if (p->data != q->data) return false;
        p = p->next;
        q = q->prior;
    }
    return true;
}
```

复杂度：时间 `O(n)`，空间 `O(1)`。

### 12 将循环单链表 `h2` 链接到循环单链表 `h1` 之后

设二者均带头结点。

```cpp
void Connect(LinkList h1, LinkList h2) {
    if (h2->next == h2) return;

    LNode *p = h1;
    while (p->next != h1) p = p->next;

    LNode *q = h2;
    while (q->next != h2) q = q->next;

    p->next = h2->next;
    q->next = h1;
}
```

复杂度：时间 `O(m+n)`，空间 `O(1)`。

### 13 按访问频度调整非循环双链表的 `Locate(L,x)`

```cpp
struct FNode {
    ElemType data;
    int freq;
    FNode *prior;
    FNode *next;
};

using FLinkList = FNode*;

FNode* Locate(FLinkList L, ElemType x) {
    FNode *p = L->next;
    while (p != nullptr && p->data != x) {
        p = p->next;
    }
    if (p == nullptr) return nullptr;

    p->freq++;

    p->prior->next = p->next;
    if (p->next != nullptr) {
        p->next->prior = p->prior;
    }

    FNode *pre = L;
    while (pre->next != nullptr && pre->next->freq > p->freq) {
        pre = pre->next;
    }

    p->next = pre->next;
    if (pre->next != nullptr) {
        pre->next->prior = p;
    }
    pre->next = p;
    p->prior = pre;

    return p;
}
```

复杂度：时间 `O(n)`，空间 `O(1)`。

### 14 不带头结点单链表循环右移 `k` 位

```cpp
void RightShift(LinkList &L, int n, int k) {
    if (L == nullptr || k == 0) return;

    LNode *tail = L;
    while (tail->next != nullptr) {
        tail = tail->next;
    }
    tail->next = L;

    LNode *newTail = L;
    for (int i = 1; i < n - k; i++) {
        newTail = newTail->next;
    }

    L = newTail->next;
    newTail->next = nullptr;
}
```

复杂度：时间 `O(n)`，空间 `O(1)`。

### 15 判断单链表是否有环

采用快慢指针。

```cpp
bool HasCycle(LinkList L) {
    LNode *slow = L;
    LNode *fast = L;

    while (fast != nullptr && fast->next != nullptr) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) return true;
    }
    return false;
}
```

复杂度：时间 `O(n)`，空间 `O(1)`。

### 16 求不带头结点单链表的最大孪生和

`n` 为偶数。先原地逆置前半段，再与后半段同步扫描。

```cpp
int MaxTwinSum(LinkList L) {
    LNode *slow = L;
    LNode *fast = L;
    LNode *pre = nullptr;

    while (fast != nullptr && fast->next != nullptr) {
        fast = fast->next->next;
        LNode *next = slow->next;
        slow->next = pre;
        pre = slow;
        slow = next;
    }

    int ans = 0;
    while (pre != nullptr && slow != nullptr) {
        int sum = pre->data + slow->data;
        if (sum > ans) ans = sum;
        pre = pre->next;
        slow = slow->next;
    }
    return ans;
}
```

复杂度：时间 `O(n)`，空间 `O(1)`。

### 17 查找倒数第 `k` 个位置上的结点

两个指针间隔 `k` 个结点同步移动。

```cpp
int SearchK(LinkList list, int k) {
    if (k <= 0) return 0;

    LNode *p = list->next;
    LNode *q = list->next;

    for (int i = 0; i < k; i++) {
        if (q == nullptr) return 0;
        q = q->next;
    }

    while (q != nullptr) {
        p = p->next;
        q = q->next;
    }

    cout << p->data << endl;
    return 1;
}
```

复杂度：时间 `O(n)`，空间 `O(1)`。

### 18 查找两个单词链表的公共后缀起始位置

公共后缀的结点地址相同。先对齐长度，再同步前进。

```cpp
int Length(LinkList L) {
    int len = 0;
    for (LNode *p = L->next; p != nullptr; p = p->next) {
        len++;
    }
    return len;
}

LNode* FindCommonSuffix(LinkList str1, LinkList str2) {
    int len1 = Length(str1);
    int len2 = Length(str2);
    LNode *p = str1->next;
    LNode *q = str2->next;

    while (len1 > len2) {
        p = p->next;
        len1--;
    }
    while (len2 > len1) {
        q = q->next;
        len2--;
    }

    while (p != nullptr && q != nullptr && p != q) {
        p = p->next;
        q = q->next;
    }
    return p;
}
```

复杂度：时间 `O(m+n)`，空间 `O(1)`。

### 19 删除绝对值重复的结点，只保留第一次出现者

利用 `|data| <= n`，开辅助数组记录某个绝对值是否出现过。

```cpp
struct Node {
    int data;
    Node *link;
};

void DeleteAbsDuplicate(Node *head, int n) {
    bool *vis = new bool[n + 1]();
    Node *pre = head;
    Node *p = head->link;

    while (p != nullptr) {
        int v = p->data >= 0 ? p->data : -p->data;
        if (vis[v]) {
            pre->link = p->link;
            delete p;
            p = pre->link;
        } else {
            vis[v] = true;
            pre = p;
            p = p->link;
        }
    }
    delete[] vis;
}
```

复杂度：时间 `O(m)`，空间 `O(n)`。

### 20 将 `L=(a1,a2,...,an)` 重排为 `L'=(a1,an,a2,an-1,...)`

步骤：

1. 用快慢指针找到中点。
2. 将后半段逆置。
3. 将前半段和逆置后的后半段交替合并。

```cpp
void Reorder(LinkList L) {
    if (L->next == nullptr || L->next->next == nullptr) return;

    LNode *slow = L->next;
    LNode *fast = L->next;
    while (fast->next != nullptr && fast->next->next != nullptr) {
        slow = slow->next;
        fast = fast->next->next;
    }

    LNode *p = slow->next;
    slow->next = nullptr;

    LNode *second = nullptr;
    while (p != nullptr) {
        LNode *q = p->next;
        p->next = second;
        second = p;
        p = q;
    }

    LNode *first = L->next;
    while (second != nullptr) {
        LNode *q1 = first->next;
        LNode *q2 = second->next;

        first->next = second;
        second->next = q1;

        first = q1;
        second = q2;
    }
}
```

复杂度：时间 `O(n)`，空间 `O(1)`。
